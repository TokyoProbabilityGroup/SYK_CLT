\title{A Central Limit Theorem for High-Temperature SYK
Model}
\author{}
\date{}

\begin{document}

\begin{abstract}
Fix an even integer $q\geq4$, and let $Z_N(\beta)$ be the
partition function of the Gaussian $q$-body Sachdev--Ye--Kitaev model with $N$
Majorana fermions.  We prove that for all sufficiently small inverse temperatures
$\beta > 0$, there exists $\rho=\rho(\beta)>0$ such that   
\[
N^{q/2 -1}(\log Z_N(\beta)-\E \log Z_N(\beta))
  \Longrightarrow \N(0,\rho^2).
\]  
\end{abstract}

\noindent\emph{Keywords:} Sachdev--Ye--Kitaev model,
free energy, central limit theorem, Gaussian chaos.
