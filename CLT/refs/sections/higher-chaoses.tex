\section{Direct Rademacher control of all higher chaoses}
\label{sec:Rademacher-remainder}
}

\begin{lemma}
\label{lem:Z-derivative-bound}
For any $\beta\in \mathbb R$ and for any multi-index $\alpha$  with  $|\alpha|=r$, it holds 
\begin{equation}\label{eq:Z-derivative-bound}
  \left|\partial_g^\alpha Z_N(\beta;G)\right|
  \leq (|\beta|\sigma_N)^r Z_N(\beta;G).
\end{equation}
\end{lemma}

\begin{proof}
The case $r=0$ is immediate, so assume $r\geq1$.  Write $H_N:=H_N(G)$ for
brevity.  We claim that 
\begin{align}
	&\partial_{g_{I_r}}\cdots\partial_{g_{I_1}}Z_N(\beta;G)
	\notag\\
	&\quad=(-\beta\sigma_N)^r
	\sum_{\pi\in S_r}\int_{\Delta_r}
	\tr\!\left(
	e^{-t_0\beta H_N}\psi_{I_{\pi(1)}}e^{-t_1\beta H_N}
	\cdots
	\psi_{I_{\pi(r)}}e^{-t_r\beta H_N}
	\right)\,\mathrm dt_1\cdots\mathrm dt_r
	\label{eq:iterated-Duhamel-formula}
\end{align}
where
\[\Delta_r=\left\{(t_j)_{j=1}^r: t_j\ge0,\quad \sum_{j=1}^r t_j\le 1\right\},\qquad t_0=1-\sum_{j=1}^r t_j. \]
For an affine matrix-valued function $A(\theta)$, the one-variable
Duhamel formula (see \cite[Corollary~3.11]{HiaiPetz2014}) is
\begin{equation}\label{eq:one-variable-Duhamel-formula}
  \frac{\mathrm d}{\mathrm d\theta}e^{A(\theta)}
  =\int_0^1
    e^{sA(\theta)}A'(\theta)e^{(1-s)A(\theta)}\,\mathrm ds.
\end{equation}
Since $\partial_{g_I}H_N=\sigma_N\psi_I$ and the fact that  $A=-t\beta H_N$ is affine in each $g_I$, by 
\eqref{eq:one-variable-Duhamel-formula},  for $t\geq0$,
\begin{align}
  \partial_{g_I}e^{-t\beta H_N}
  &=-t\beta\sigma_N\int_0^1
    e^{-st\beta H_N}\psi_I
    e^{-(1-s)t\beta H_N}\,\mathrm ds \notag\\
  &=-\beta\sigma_N\int_0^t
    e^{-u\beta H_N}\psi_I
    e^{-(t-u)\beta H_N}\,\mathrm du.
  \label{eq:Duhamel-interval-splitting}
\end{align}
Note that  \eqref{eq:Duhamel-interval-splitting} with $t=1$ yields
\eqref{eq:iterated-Duhamel-formula} for $r=1$.  Suppose inductively that the
formula holds at order $r-1$.  Then
\begin{align}
	&\partial_{g_{I_r}}\cdots\partial_{g_{I_1}}Z_N(\beta;G)
	\notag\\
	&\quad=(-\beta\sigma_N)^{r-1}
	\sum_{\pi\in S_{r-1}}\int_{\Delta_{r-1}}\partial_{g_{I_r}}
	\tr\!\left(
	e^{-t_0\beta H_N}\psi_{I_{\pi(1)}}e^{-t_1\beta H_N}
	\cdots
	\psi_{I_{\pi(r-1)}}e^{-t_{r-1}\beta H_N}
	\right)\,\mathrm dt_1\cdots\mathrm dt_{r-1}.
	\label{eq:iterated-Duhamel-formula-induction}
\end{align}
For fixed $\pi\in S_{r-1}$ and $j\in\{1,\ldots,r\}$, by
\eqref{eq:Duhamel-interval-splitting}, differentiating the
factor $e^{-t_{j-1}\beta H_N}$ replaces it  with
\[
  -\beta\sigma_N\int_0^{t_{j-1}}
  e^{-u\beta H_N}\psi_{I_r}
  e^{-(t_{j-1}-u)\beta H_N}\,\mathrm du.
\]
Splitting the interval $[0, t_{j-1}]$ into $[0, u]$ and $[u, t_{j-1}]$ and relabeling the resulting
simplex variables, the corresponding term in the Leibniz rule becomes
\begin{align*}
  -\beta\sigma_N\int_{\Delta_r}
  \tr\!\left(
    e^{-t_0\beta H_N}\psi_{I_{\sigma_{\pi,j}(1)}}e^{-t_1\beta H_N}
    \cdots
    \psi_{I_{\sigma_{\pi,j}(r)}}e^{-t_r\beta H_N}
  \right)\,\mathrm dt_1\cdots\mathrm dt_r,
\end{align*}
where
\[\sigma_{\pi,j}(k)=\begin{cases}
	\pi(k) & (k<j),\\
	r &(k=j),\\
	\pi(k-1) & (k>j).
\end{cases} \]
Since $(\pi,j)\mapsto\sigma_{\pi,j}$ is a bijection from
$S_{r-1}\times\{1,\ldots,r\}$ onto $S_r$, summing the terms produced by the Leibniz rule in
\eqref{eq:iterated-Duhamel-formula-induction} proves
\eqref{eq:iterated-Duhamel-formula}.

To bound the integrand in
\eqref{eq:iterated-Duhamel-formula}, use the normalized  Schatten norms:
\[
  \norm{A}_p=(\tr|A|^p)^{1/p},
  \qquad
  \norm{A}_\infty=\norm{A}.
\]
For these normalized Schatten norms, H\"older's inequality reads
\begin{equation}\label{eq:normalized-Schatten-Holder}
  \norm{AB}_p\leq\norm{A}_{p_1}\norm{B}_{p_2},
  \qquad
  \frac1p=\frac1{p_1}+\frac1{p_2},
  \qquad p,p_1,p_2\in[1,\infty].
\end{equation}
This follows from the inequality 
in \cite[Theorem~6.20]{HiaiPetz2014}, since, in dimension $d$,
\[
  \norm{A}_s=d^{-1/s}(\operatorname{Tr}|A|^s)^{1/s}
  \quad (1\leq s<\infty),
  \qquad d^{-1/p}=d^{-1/p_1}d^{-1/p_2},
\]
with the convention \(d^{-1/\infty}=1\).  
Let $t_0,\ldots,t_r\geq0$ satisfy $\sum_{j=0}^rt_j=1$.  Iterating \eqref{eq:normalized-Schatten-Holder} gives
\begin{align*}
  |\tr(A_0\cdots A_r)|
  &\leq\norm{A_0\cdots A_r}_1\\
  &\leq\norm{A_0\cdots A_{r-1}}_{1/(t_0+\cdots+t_{r-1})}
       \norm{A_r}_{1/t_r}.
\end{align*}
Applying this inductively, this is bounded from above by 
$$\prod_{j=0}^r\norm{A_j}_{1/t_j}.$$

In particular, since $\norm{AB}_p\leq\norm{A}_p\norm{B}_\infty$,
\[
  |\tr(A_0B_1A_1\cdots B_rA_r)|
  \leq\prod_{j=0}^r\norm{A_j}_{1/t_j}
       \prod_{k=1}^r\norm{B_k}_\infty.
\]

We apply this inequality with $A_j=e^{-t_j\beta H_N}$ and
$B_k=\psi_{I_{\pi(k)}}$.  Since $H_N$ is self-adjoint, for $t_j>0$,
\[
  \norm{e^{-t_j\beta H_N}}_{1/t_j}
  =\left(\tr\left|e^{-t_j\beta H_N}\right|^{1/t_j}\right)^{t_j}
  =
  \bigl(\tr e^{-\beta H_N}\bigr)^{t_j},
\]
For $t_j=0$, the corresponding identity is
$\norm{I}_\infty=1=(\tr e^{-\beta H_N})^0$.
Also, $\norm{\psi_I}_\infty=1$ (see Subsection~\ref{subsec:SYK-model}).
Thus, for every $\pi\in S_r$ and $t\in\Delta_r$,
\begin{align}
  &\left|
  \tr\!\left(
    e^{-t_0\beta H_N}\psi_{I_{\pi(1)}}e^{-t_1\beta H_N}
    \cdots
    \psi_{I_{\pi(r)}}e^{-t_r\beta H_N}
  \right)
  \right| \notag\\
  &\quad\leq
  \prod_{j=0}^r\norm{e^{-t_j\beta H_N}}_{1/t_j}
  \prod_{k=1}^r\norm{\psi_{I_{\pi(k)}}}_\infty \notag\\
  &\quad\leq
  \prod_{j=0}^r
  \left(\tr e^{-\beta H_N}\right)^{t_j}
  =\tr e^{-\beta H_N}
  =Z_N(\beta;G),
  \label{eq:Duhamel-integrand-Holder-bound}
\end{align}
where we used $\sum_{j=0}^rt_j=1$.  
Combining \eqref{eq:iterated-Duhamel-formula} and 
\eqref{eq:Duhamel-integrand-Holder-bound}, we obtain
\begin{align*}
 \left|
  \partial_{g_{I_r}}\cdots\partial_{g_{I_1}}Z_N(\beta;G)
  \right|
  &\leq
  (|\beta|\sigma_N)^r
  \sum_{\pi\in S_r}\int_{\Delta_r}Z_N(\beta;G)
  \,\mathrm dt_1\cdots\mathrm dt_r\\
  &=
  (|\beta|\sigma_N)^rZ_N(\beta;G)|S_r|\int_{\Delta_r}\mathrm dt_1\cdots\mathrm dt_r\\
  &=
  (|\beta|\sigma_N)^rZ_N(\beta;G),
\end{align*}
which proves \eqref{eq:Z-derivative-bound}.
\end{proof}

Recall that Lemma~\ref{lem: parity rule lemma} shows that
only even multi-indices have nonzero Hermite coefficients.

Let $(\varepsilon_x)_{x=1}^N$ be independent Rademacher variables on an
auxiliary probability space, and write $\E_\varepsilon$ for expectation on
this space.  Set
\begin{equation}\label{eq:Rademacher-chaos-definition}
  S_{N,q}
  :=\sum_{I\in\calI_N}\varepsilon_I,
  \qquad
  \varepsilon_I:=\prod_{x\in I}\varepsilon_x.
\end{equation}
Thus $S_{N,q}$ is the degree-$q$ elementary symmetric polynomial in the
Rademacher signs.

\begin{lemma}
\label{lem:Rademacher-encoding}
For every integer $r\geq0$,
\begin{equation}\label{eq:even-sum-Rademacher-moment}
  \sum_{\substack{|\alpha|=r\\ \alpha\ \mathrm{even}}}
  \frac1{\alpha!}
  =\frac1{r!}\E_\varepsilon\!\left[S_{N,q}^r\right].
\end{equation}
Moreover, there are constants $c_q>0$ and $C_q<\infty$, depending only on
$q$, such that for every $N$ and every $0\leq t\leq c_qN^{-(q-1)}$, 
\begin{equation}\label{eq:Rademacher-exponential-moment}
  \E_\varepsilon\!\left[
    |S_{N,q}|^3e^{t|S_{N,q}|}
  \right]
  \leq C_qN^{3q/2}.
\end{equation}

\end{lemma}

\begin{proof}
Starting from the definition \eqref{eq:Rademacher-chaos-definition} and
applying the multinomial formula gives
\begin{equation}\label{eq:Rademacher-multinomial-expansion}
  \frac1{r!}\E_\varepsilon\!\left[S_{N,q}^r\right]
  =\frac1{r!}\E_\varepsilon\!\left[
    \sum_{|\alpha|=r}r!
    \prod_{I\in\calI_N}\frac{\varepsilon_I^{\alpha_I}}{\alpha_I!}
  \right]
  =
  \sum_{|\alpha|=r}\frac1{\alpha!}
  \E_\varepsilon\!\left[
    \prod_{I\in\calI_N}\varepsilon_I^{\alpha_I}
  \right].
\end{equation}
For a fixed $\alpha$, the product inside the expectation is
\begin{equation*}
  \prod_{x=1}^N
  \varepsilon_x^{\sum_{I: x\in I}\alpha_I}.
\end{equation*}
By independence of the Rademacher variables and the definition of an even
multi-index in \eqref{eq:even-multi-index}, its expectation is one if
$\alpha$ is even and zero otherwise.  Substituting this into
\eqref{eq:Rademacher-multinomial-expansion} proves
\eqref{eq:even-sum-Rademacher-moment}.

We next bound $S_{N,q}$.  For every $N\geq1$
and every integer $m\geq0$, let $e_m^{(N)}$ denote the $m$th elementary
symmetric polynomial in $N$ variables, that is,
\begin{equation*}
  e_m^{(N)}(z_1,\ldots,z_N)
  :=\sum_{\substack{I\subset[N]\\ |I|=m}}
  \prod_{x\in I}z_x,
  \qquad e_0^{(N)}(z_1,\ldots,z_N)=1.
\end{equation*}
For the rest of the proof, we abbreviate 
\[
  e_m:=e_m^{(N)}(\varepsilon_1,\ldots,\varepsilon_N).
\]
Then, $e_1=\sum_{x=1}^N\varepsilon_x$, and
\eqref{eq:Rademacher-chaos-definition} gives $S_{N,q}=e_q$.
Newton's identities state that
\begin{equation}\label{eq:Newton-identities}
  m e_m
  =\sum_{j=1}^m(-1)^{j-1}e_{m-j}p_j,
\end{equation}
where $p_j:=\sum_{x=1}^N\varepsilon_x^j$.  Since
$\varepsilon_x\in\{\pm1\}$, we have
\begin{equation*}
  p_j
  =
  \begin{cases}
    e_1,&j\text{ odd},\\
    N,&j\text{ even}.
  \end{cases}
\end{equation*}
We prove this by induction on $m$ to  shows that there is a constant $C_m<\infty$ such that
\begin{equation*}
	|e_m|
	\le C_m\left(N^{m/2}+|e_1|^m\right).
\end{equation*}
Indeed, suppose that the estimate holds for  orders less than $m$. Set
\[
X_N:=\frac{e_1}{\sqrt N}.
\]
For every $j\ge1$,
\[
|p_j|\le N^{j/2}(1+|X_N|^j).
\]
Hence, with $C_0=1$,
\[
\begin{aligned}
	|e_{m-j}p_j|
	&\le
	C_{m-j}N^{m/2}
	(1+|X_N|^{m-j})(1+|X_N|^j)\\
	&\le
	2C_{m-j}N^{m/2}(1+|X_N|^m),
\end{aligned}
\]
where we used $1+|X|^m-|X|^{m-j}-|X|^j
=(1-|X|^j)(1-|X|^{m-j})\ge0.$
Substituting this into \eqref{eq:Newton-identities} proves the induction step.
Taking $m=q$ and using \eqref{eq:Rademacher-chaos-definition}, we obtain
\begin{equation}\label{eq:Rademacher-chaos-deterministic-bound}
  |S_{N,q}|
  \leq C_q\left(N^{q/2}+|e_1|^q\right).
\end{equation}

Since $q\geq4$ and
$|X_N|\leq\sqrt N$,
\eqref{eq:Rademacher-chaos-deterministic-bound} implies that,
for $0\leq t\leq c_qN^{-(q-1)}$,
\begin{align}
  t|S_{N,q}|
  &\leq C_qc_qN^{-(q/2-1)}\left(1+|X_N|^q\right)\leq C_qc_q\left(1+|X_N|^2\right).
  \label{eq:Rademacher-exponent-reduction}
\end{align}
Indeed,
\begin{equation*}
  N^{-(q/2-1)}|X_N|^q
  =|X_N|^2
  \left(\frac{|X_N|}{\sqrt N}\right)^{q-2}
  \leq |X_N|^2.
\end{equation*}

Choose $c_q$ small enough so that $C_q c_q\leq 1/4$.  Hoeffding's inequality~\cite[Theorem~2]{Hoeffding1963} gives
\begin{equation}\label{eq:Hoeffding-Rademacher-sum}
  \mathbb P_\varepsilon(|X_N|\geq u)
  \leq2e^{-u^2/2}
  \qquad (u\geq0).
\end{equation}
Consequently, integrating the tail bound   in
\eqref{eq:Hoeffding-Rademacher-sum} yields
\begin{equation}\label{eq:uniform-Rademacher-integrability}
  \sup_{N\geq1}
  \E_\varepsilon\!\left[
    \left(1+|X_N|^{3q}\right)e^{|X_N|^2/4}
  \right]
  <\infty.
\end{equation}
Finally, \eqref{eq:Rademacher-chaos-deterministic-bound},
\eqref{eq:Rademacher-exponent-reduction}, and
\eqref{eq:uniform-Rademacher-integrability} give
\begin{align*}
  \E_\varepsilon\!\left[
    |S_{N,q}|^3e^{t|S_{N,q}|}
  \right]
  &\leq
  C_qN^{3q/2}
  \E_\varepsilon\!\left[
    \left(1+|X_N|^{3q}\right)
    e^{(1+|X_N|^2)/4}
  \right] \\
  &\leq C_qN^{3q/2},
\end{align*}
which proves \eqref{eq:Rademacher-exponential-moment}.
\end{proof}

\begin{proposition} \label{prop:higher-chaos-closure}
There exists $\beta_{\rm r}=\beta_{\rm r}(q)>0$ such that, for every
fixed real $\beta$ with $|\beta|<\beta_{\rm r}$,
\begin{equation*}
  \sum_{r\geq3}\norm{P_r Y_N(\beta)}_2^2
  =O\!\left(N^{-3(q/2-1)}\right)
  =o(N^{2-q}).
\end{equation*}
\end{proposition}

\begin{proof}
Put
\begin{equation*}
  t_N:=\beta^2\sigma_N^2
  =\beta^2(q-1)!N^{-(q-1)}.
\end{equation*}
By the Hermite expansion \eqref{eq: chaos} and its orthogonality relation,
\[
  \norm{P_r Y_N(\beta)}_2^2
  =\frac1{(\E Z_N(\beta))^2}
    \sum_{|\alpha|=r}\frac1{\alpha!}
    \left|\E\!\left[Z_N(\beta)H_\alpha(G)\right]\right|^2.
\]
By Lemmas~\ref{lem: parity rule lemma} and~\ref{lem:Gaussian-Hermite-IBP},
\[
  \E\!\left[Z_N(\beta)H_\alpha(G)\right]
  =\begin{cases}
    0,&\alpha\text{ is odd},\\
    \E\partial_g^\alpha Z_N(\beta),&\alpha\text{ is even}.
  \end{cases}
\]
Moreover, Lemma~\ref{lem:Z-derivative-bound} gives, for $|\alpha|=r$,
\[
  \left|\E\partial_g^\alpha Z_N(\beta)\right|
  \leq\E\left|\partial_g^\alpha Z_N(\beta)\right|
  \leq (|\beta|\sigma_N)^r\E Z_N(\beta)
  =t_N^{r/2}\E Z_N(\beta).
\]
Since $\E Z_N(\beta)>0$, it follows that
\begin{align}
  \norm{P_r Y_N(\beta)}_2^2
  &=\frac1{(\E Z_N(\beta))^2}
    \sum_{\substack{|\alpha|=r\\ \alpha\ \mathrm{even}}}
    \frac1{\alpha!}
    \left|\E\partial_g^\alpha Z_N(\beta)\right|^2 \notag\\
  &\leq t_N^r
    \sum_{\substack{|\alpha|=r\\ \alpha\ \mathrm{even}}}
    \frac1{\alpha!}.
  \label{eq:Hermite-Rademacher-bound}
\end{align}
By Lemma~\ref{lem:Rademacher-encoding}, summing
\eqref{eq:Hermite-Rademacher-bound} over all $r\geq3$ yields
\begin{align}
  \sum_{r\geq3}\norm{P_r Y_N(\beta)}_2^2
  &\leq
  \sum_{r\geq3}\frac{t_N^r}{r!}
  \E_\varepsilon\!\left[S_{N,q}^r\right] \notag\\
  &\leq
  \E_\varepsilon\!\left[
    \sum_{r\geq3}\frac{(t_N|S_{N,q}|)^r}{r!}
  \right] \notag\\
  &\leq
  \frac{t_N^3}{6}
  \E_\varepsilon\!\left[
    |S_{N,q}|^3e^{t_N|S_{N,q}|}
  \right],
  \label{eq:all-higher-chaoses-at-once}
\end{align}
where the second inequality uses
$|\E_\varepsilon[S_{N,q}^r]|\leq
\E_\varepsilon[|S_{N,q}|^r]$, and the last inequality uses
$\sum_{r\geq3}u^r/r!\leq u^3e^u/6$ for $u\geq0$.
Choose
\begin{equation*}
  \beta_{\rm r}
  :=\sqrt{\frac{c_q}{(q-1)!}},
\end{equation*}
where $c_q$ is the constant in Lemma~\ref{lem:Rademacher-encoding}.  If
$|\beta|<\beta_{\rm r}$, then
$t_N\leq c_qN^{-(q-1)}$.  Therefore the second claim of
Lemma~\ref{lem:Rademacher-encoding}, together with
\eqref{eq:all-higher-chaoses-at-once} implies 
\begin{align*}
  \sum_{r\geq3}\norm{P_r Y_N(\beta)}_2^2
  &\leq C_{q,\beta}
  N^{-3(q-1)}N^{3q/2}=C_{q,\beta}N^{-3(q/2-1)}.
\end{align*}
Since $q\geq4$, we have $3(q/2-1)>q-2$, which completes the proof.
\end{proof}
