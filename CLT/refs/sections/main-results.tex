\section{Proof of the main results}
\label{sec:main-results}

To identify the deterministic coefficient $A_N(\beta)$ in
\eqref{eq:finite-N-FTW-coefficient}, we use the annealed free energy limit
from \cite{He2026}, which holds at every fixed non-negative $\beta$. 
The symmetry of gaussian distributions extends the claim to negative \(\beta\).

The following is a standard convergence theorem for convex functions, \textcolor{red}{so we omit its proof}; see
\cite[Theorems~10.8 and~25.7]{Rockafellar1970}.

\begin{lemma} \label{lem:convex-derivatives}
Let $h\in C^1(\mathbb R )$. Suppose that $h_N:\mathbb R \to\mathbb R$ are
 differentiable, convex functions, and that $h_N\to h$ pointwise. Then locally uniformly on $\mathbb R $, 
\[
    h_N\to h
    \qquad\text{and}\quad
    h_N'\to h'.
\]

\end{lemma}
\textcolor{red}{The next proposition establishes convergence of the annealed free energy and its derivative.}
\begin{proposition} \label{prop:annealed-free-energy}
    There exists an even convex function $\ell_q\in C^1(\mathbb R)$ such that,
    locally uniformly for $\beta\in\mathbb R$,
    \begin{align}
        \frac1N\log \E[Z_N(\beta)]
        &\longrightarrow \ell_q(\beta),
        \label{eq:annealed-free-energy-limit}\\
        \frac1N\partial_\beta\log \E[Z_N(\beta)]
        &\longrightarrow \ell_q'(\beta).
        \label{eq:annealed-derivative-limit}
    \end{align}
    Moreover, $\beta\ell_q'(\beta)>0$ for every $\beta\neq0$.
\end{proposition}

\begin{proof}
    Set
    \begin{equation*}
        \widetilde{H}_N:=2^{-q/2}H_N,
        \qquad
       \textcolor{red}{\widetilde{p}_{N,q}(s)}:=\frac1N\log\E[\tr e^{s\widetilde{H}_N}].
    \end{equation*}
    Since $G$ and $-G$ have the
    same law and $H_N(-G)=-H_N(G)$, for every $\beta\in\mathbb R$,
    \begin{equation} \label{eq:He-normalization-conversion}
        \frac1N\log\E[Z_N(\beta)]
        =\frac1N\log\E[\tr e^{|\beta|H_N}]
        =\widetilde{p}_{N,q}(2^{q/2}|\beta|).
    \end{equation}
    Writing $p_q:=p_{{\rm SD},q}$ for the Schwinger--Dyson pressure, \cite[Theorem~1.1 and Proposition~3.3]{He2026} give
    \begin{equation*}
        \widetilde{p}_{N,q}(s)\longrightarrow p_q(s)
        \quad (s>0),
        \qquad
        p_q\in C^1((0,\infty)),
        \qquad
        \lim_{s\downarrow0}p_q(s)=0.
    \end{equation*}
    Define
    \begin{equation*}
        \ell_q(\beta):=
        \begin{cases}
            p_q(2^{q/2}|\beta|),&\beta\neq0,\\
            0,&\beta=0.
        \end{cases}
    \end{equation*}
    Then $\ell_q$ is even and continuous on $\mathbb R$. By \eqref{eq:He-normalization-conversion} and  $Z_N(0)=1$, we have pointwise convergence in \eqref{eq:annealed-free-energy-limit}.

    By the derivative formula in \cite[Proposition 3.3]{He2026}, for any $s>0$, 
    \begin{equation} \label{eq:He-pressure-derivative}
        p_q'(s)=\frac{s}{q}\int_0^1G_{s,q}(u)^q\,\mathrm du.
    \end{equation}
    By the positive Lehmann representation and
    \cite[Theorem~2.5]{He2026}, there is a probability measure $\nu_s$
    on $[0,\infty)$ such that
    \begin{equation*}
        G_{s,q}(u)
        =\int_0^\infty
        \frac{\cosh(x(1/2-u))}{2\cosh(x/2)}\,\nu_s(\mathrm dx).
    \end{equation*}
    For $\beta>0$, put
    \begin{equation*}
        K_\beta(u):=2G_{2^{q/2}\beta,q}(u).
    \end{equation*}
    The chain rule and
    \eqref{eq:He-pressure-derivative} yield
    \begin{equation} \label{eq:ell-kernel-derivative}
        \ell_q'(\beta)
        =2^{q/2}p_q'(2^{q/2}\beta)
        =\frac{2^q\beta}{q}\int_0^1G_{2^{q/2}\beta,q}(u)^q\,\mathrm du
        =\frac\beta q\int_0^1K_\beta(u)^q\,\mathrm du.
    \end{equation}
    Moreover,
    \begin{equation} \label{eq:ell-kernel-representation}
        K_\beta(u)
        =\int_0^\infty
        \frac{\cosh(x(1/2-u))}{\cosh(x/2)}\,\nu_{2^{q/2}\beta}(\mathrm dx).
    \end{equation}

    For $x\geq0$ and $u\in[0,1]$,
    \begin{equation*}
        0<\frac{\cosh(x(1/2-u))}{\cosh(x/2)}\leq1.
    \end{equation*}
    Since $\nu_{2^{q/2}\beta}$ is a probability measure,
    \eqref{eq:ell-kernel-representation} gives $0<K_\beta(u)\leq1$.
    Hence \eqref{eq:ell-kernel-derivative} implies
    \begin{equation*}
        0<\ell_q'(\beta)\leq\frac\beta q, 
        \qquad \beta>0.
    \end{equation*}
    Integrating over $[0,\beta]$ with $\ell_q(0)=0$ 
    \textcolor{red}{gives} $0\leq\ell_q(\beta)\leq\beta^2/(2q)$ for $\beta>0$.
    Since $\ell_q$ is even, this bound holds for every $\beta\in\mathbb R$, thus 
    \begin{equation*}
        \ell_q'(0)= \lim_{\beta\to0}
        \frac{\ell_q(\beta)-\ell_q(0)}{\beta}=0.
    \end{equation*}

On $\mathbb R\setminus\{0\}$, the derivative
    $\ell_q'$ is continuous by $p_q\in C^1((0,\infty))$ and is odd
    by evenness of $\ell_q$. Therefore
    $|\ell_q'(\beta)|\leq|\beta|/q$, which gives continuity at zero
    and hence $\ell_q\in C^1(\mathbb R)$.
    Strict positivity for $\beta>0$ and oddness also give
    $\beta\ell_q'(\beta)>0$ for every $\beta\neq0$.

It remains to prove convexity. 
    Set
    \begin{equation*}
        F_N(\beta):=\E[Z_N(\beta)],
        \qquad
        h_N(\beta):=\frac1N\log F_N(\beta),
        \qquad
        \overline\nu_N(B):=\E[\tr\mathbf1_B(H_N)].
    \end{equation*}
    Then $\overline\nu_N$ is a probability measure and
    \begin{equation*}
        F_N(\beta)=\int_{\mathbb R}e^{-\beta x}\,
        \overline\nu_N(\mathrm dx).
    \end{equation*}
 In particular,
    $h_N$ is finite and twice continuously differentiable.  With the
    probability measure
    \begin{equation*}
        \overline\nu_{N,\beta}(\mathrm dx)
        :=\frac{e^{-\beta x}}{F_N(\beta)}\,
        \overline\nu_N(\mathrm dx),
    \end{equation*}
    we have
    \begin{equation*}
        h_N''(\beta)
        =\frac1N\left(
            \int_{\mathbb R}x^2\,\overline\nu_{N,\beta}(\mathrm dx)
            -\left(\int_{\mathbb R}x\,
            \overline\nu_{N,\beta}(\mathrm dx)\right)^2
        \right)\geq0.
    \end{equation*}
    Thus, $h_N$ is convex. 
    Applying Lemma~\ref{lem:convex-derivatives} proves
    both \eqref{eq:annealed-free-energy-limit} and
    \eqref{eq:annealed-derivative-limit} locally uniformly on $\mathbb R$.
\end{proof}


By \eqref{eq:finite-N-FTW-coefficient} and
\eqref{eq:annealed-derivative-limit}, locally uniformly for
$\beta\in\mathbb R$,
\begin{equation*}
  \frac{A_N(\beta)}N\longrightarrow
  \frac\beta2\ell_q'(\beta).
\end{equation*}
\textcolor{red}{We are now in a position to prove Theorems~\ref{thm:partition-clt} and~\ref{thm:free-energy-clt}.}

\begin{proof}[Proof of Theorem~\ref{thm:partition-clt}]
Note that $\beta\ell_q'(\beta)>0$ from Proposition~\ref{prop:annealed-free-energy}. Let
\[
  U_N:=\frac1{\sqrt{2|\mathcal I_N|}}\sum_{I\in\calI_N}(g_I^2-1).
\]
The classical i.i.d.\ central limit theorem gives
$U_N\Rightarrow\N(0,1)$.
By Lemma~\ref{lem:exact-second-chaos},
\begin{equation}\label{eq:Y-decomposition-final}
  Y_N(\beta)-1
  =c_N(\beta)U_N+\mathcal R_N(\beta),
  \qquad
  c_N(\beta)
  :=\frac{\beta\,\partial_\beta\log \E[Z_N(\beta)]}{\sqrt{2|\mathcal I_N|}},
\end{equation}
where $\mathcal R_N:=\sum_{r\geq3}P_r Y_N$ is orthogonal to $U_N$.
Since $|\mathcal I_N|=\binom Nq\sim N^q/q!$, \eqref{eq:annealed-derivative-limit} gives
\[
  N^{q/2-1}c_N(\beta)
  \longrightarrow
  \sqrt{\frac{q!}{2}}\,\beta\ell_q'(\beta).
\]
Proposition~\ref{prop:higher-chaos-closure} gives
$N^{q/2-1}\mathcal R_N\to0$ in $L^2$.  This proves
\eqref{eq:scaled-relative-clt}.  Orthogonality in
\eqref{eq:Y-decomposition-final} also gives
\[
  \Var(Y_N)=c_N(\beta)^2+\norm{\mathcal R_N(\beta)}_2^2.
\]
Finally, using the identity 
$\Var(Y_N)=\Var(Z_N)/(\E[Z_N])^2$, this  
 gives \eqref{eq:standardized-clt}.
\end{proof}

\begin{proof}[Proof of \textcolor{red}{Theorem~\ref{thm:free-energy-clt}}]
Assume $\beta\neq0$.  Set
\[
a:=\frac q2-1,
\qquad
W_N:=\log Y_N(\beta).
\]
Then
\begin{align*}
N^a\bigl(\log Z_N-\E[\log Z_N]\bigr)
&=
N^a\bigl(W_N-(Y_N-1)\bigr)
+N^a(Y_N-1)
-N^a\E[W_N] .
\end{align*}

By Theorem~\ref{thm:partition-clt},
\[
N^a(Y_N-1)
\Longrightarrow
\sqrt{\frac{q!}{2}}\,
\beta\ell_q'(\beta)\,\xi,
\qquad
\xi\sim\mathcal N(0,1).
\]
Moreover, $Y_N-1\to0$ in probability and
$N^a(Y_N-1)$ is tight.  Since whenever $|x|\le1/2$, 
\[
|\log(1+x)-x|\le x^2,
\]
it follows that
\[
N^a\bigl(W_N-(Y_N-1)\bigr)
\xrightarrow{\mathrm p}0.
\]
Indeed, on $\{|Y_N-1|\leq1/2\}$ the absolute value is bounded by
\(
|N^a(Y_N-1)|\,|Y_N-1|
\), which converges to zero in probability by tightness of the first factor
and convergence of the second factor to zero.  The complementary event has
probability tending to zero.
Thus it remains to prove
\[
\E[W_N]=o(N^{-a}).
\]

Write, as in \eqref{eq:Y-decomposition-final},
\[
Y_N-1=c_NU_N+\mathcal R_N,
\]
where
\[
c_N\coloneqq \frac{\beta\,\partial_\beta\log \E[Z_N(\beta)]}{\sqrt{2|\mathcal I_N|}}= O(N^{-a}),\quad U_N\coloneqq \frac1{\sqrt{2|\mathcal I_N|}}\sum_{I\in\mathcal I_N}(g_I^2-1)
,\quad
\|\mathcal R_N\|_2^2=O(N^{-3a})
\]
by Proposition~\ref{prop:higher-chaos-closure}. Orthogonality therefore
gives
\[
\E[(Y_N-1)^2]=c_N^2 \E[U_N^2] + \E[\mathcal R_N^2]= O(N^{-2a}).
\]
Since $\E[(g_I^2-1)^2]=2$ and
$\E[(g_I^2-1)^4]=60$, independence gives
\[
\E[U_N^4]=3+\frac{12}{|\mathcal I_N|}\leq15.
\]  
Hence, we estimate
\begin{align*}
\mathbb P(Y_N<1/2)
&\le
\mathbb P(|c_NU_N|>1/4)
+
\mathbb P(|\mathcal R_N|>1/4)\\
&\le
C c_N^4\E[U_N^4]
+
C\|\mathcal R_N\|_2^2\\
&=O(N^{-3a}).
\end{align*}

We claim that 
\[
Z_N(\beta;G)\ge1.
\]
To this end, fix \(I\in\mathcal I_N\) and choose \(x\in I\). 
The unitary $U_x$ from Lemma~\ref{lem: parity rule lemma} satisfies
\[
U_x\psi_IU_x^*=-\psi_I,
\]
and hence $\tr\psi_I=0$.  Therefore
\(
\tr H_N(G)=0.
\) 
Jensen's inequality applied to the eigenvalues of $H_N(G)$ now gives
\[
Z_N(\beta;G)
=
\tr e^{-\beta H_N(G)}
\ge
e^{-\beta\tr H_N(G)}
=1.
\]

Define
\[
h(y):=y-1-\log y,
\qquad y>0.
\]
Since $\E[Y_N]=1$,
\[
-\E[W_N]
=
\E[h(Y_N)].
\]
On $\{Y_N\ge1/2\}$,
\[
h(Y_N)\le (Y_N-1)^2,
\]
and therefore
\[
\E\!\left[
h(Y_N)\mathbf 1_{\{Y_N\ge1/2\}}
\right]
=O(N^{-2a}).
\]

On the other hand, if $Y_N<1/2$, then
\[
h(Y_N)\le-\log Y_N.
\]
Since $Z_N\ge1$,
\[
-\log Y_N
=
\log\E[Z_N]-\log Z_N
\le
\log\E[Z_N].
\]
By Proposition~\ref{prop:annealed-free-energy},
\(
\log\E[Z_N]=O(N),
\) 
we have
\begin{align*}
\E\!\left[
h(Y_N)\mathbf 1_{\{Y_N<1/2\}}
\right]
&\le
O(N)\,\mathbb P(Y_N<1/2)=
O(N^{1-3a})
=O(N^{-2a}),
\end{align*}
where the last estimate uses $a=q/2-1\geq1$.
Consequently,
\[
0\le-\E[W_N]
=
O(N^{-2a})
=o(N^{-a}).
\]

Combining the three terms above and applying \textcolor{red}{Slutsky's lemma
\cite[Theorem~3.1]{Billingsley1999}} yields
\[
N^{q/2-1}
\bigl(
\log Z_N(\beta)-\E[\log Z_N(\beta)]
\bigr)
\Longrightarrow
\mathcal N\left(
0,\frac{q!}{2}\beta^2(\ell_q'(\beta))^2
\right).
\]
\end{proof}
